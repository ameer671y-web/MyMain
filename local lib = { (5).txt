local function RegisterTheme(object, property)
    if object then
        table.insert(ThemeObjects, {
            Object = object,
            Property = property
        })
        pcall(function()
            object[property] = PresetColor
        end)
    end
end

local function ApplyThemeColor(color, save)
    if typeof(color) ~= "Color3" then
        return
    end

    if save then
        ThemeColor = color
    end

    PresetColor = color

    for i = #ThemeObjects, 1, -1 do
        local data = ThemeObjects[i]
        if data.Object and data.Object.Parent then
            pcall(function()
                data.Object[data.Property] = color
            end)
        else
            table.remove(ThemeObjects, i)
        end
    end

    for i = #ToggleObjects, 1, -1 do
        local data = ToggleObjects[i]

        if data.Fill and data.Fill.Parent then
            data.Fill.BackgroundColor3 = color
        else
            table.remove(ToggleObjects, i)
        end
    end

    if ActiveTabTitle and ActiveTabTitle.Parent then
        ActiveTabTitle.TextColor3 = color
    end
end

local function MakeDraggable(topbarobject, object)
    local dragging = false
    local dragInput
    local dragStart
    local startPosition

    local function update(input)
        if not dragging then
            return
        end

        local delta = input.Position - dragStart

        object.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end

    topbarobject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then

            dragging = true
            dragStart = input.Position
            startPosition = object.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    topbarobject.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            update(input)
        end
    end)
end

local function AddCorner(object, radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius or 7)
    corner.Parent = object
    return corner
end

local function AddStroke(object, color, transparency, thickness)
    local stroke = Instance.new("UIStroke")
    stroke.Color = color or PresetColor
    stroke.Transparency = transparency or 0
    stroke.Thickness = thickness or 1
    stroke.Parent = object
    return stroke
end

function lib:ChangePresetColor(color)
    if typeof(color) == "Color3" then
        ApplyThemeColor(color, true)
    end
end

    GetKey.Font = Enum.Font.GothamBold
    GetKey.TextColor3 = Color3.new(1, 1, 1)
    GetKey.TextSize = 13
    GetKey.AutoButtonColor = false

    AddCorner(GetKey, 9)

    local Verify = Instance.new("TextButton")
    Verify.Parent = Frame
    Verify.Position = UDim2.new(0.52, 0, 0.65, 0)
    Verify.Size = UDim2.new(0.4, -5, 0, 40)
    Verify.BackgroundColor3 = PresetColor
    Verify.BorderSizePixel = 0
    Verify.Text = "Verify"
    Verify.Font = Enum.Font.GothamBold
    Verify.TextColor3 = Color3.new(1, 1, 1)
    Verify.TextSize = 13
    Verify.AutoButtonColor = false

    AddCorner(Verify, 9)

    RegisterTheme(Verify, "BackgroundColor3")

    local Status = Instance.new("TextLabel")
    Status.Parent = Frame
    Status.BackgroundTransparency = 1
    Status.Position = UDim2.new(0.08, 0, 0.84, 0)
    Status.Size = UDim2.new(0.84, 0, 0, 25)
    Status.Font = Enum.Font.Gotham
    Status.Text = ""
    Status.TextColor3 = Color3.fromRGB(220, 80, 80)
    Status.TextSize = 12
    Status.TextXAlignment = Enum.TextXAlignment.Center

    GetKey.Activated:Connect(function()
        Status.TextColor3 = Color3.fromRGB(200, 200, 200)
        Status.Text = "Key: " .. key
    end)

    Verify.Activated:Connect(function()
        if Input.Text == key then
            unlocked = true
            Status.TextColor3 = Color3.fromRGB(70, 220, 110)
            Status.Text = "Key accepted!"

            task.delay(0.4, function()
                if KeyGui.Parent then
                    KeyGui:Destroy()
                end
            end)
        else
            Status.TextColor3 = Color3.fromRGB(230, 70, 70)
            Status.Text = "Invalid key!"
        end
    end)

    return function()
        return unlocked
    end
end
