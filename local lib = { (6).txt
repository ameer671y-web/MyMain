local function RegisterTheme(object, property)
    if object then
        table.insert(ThemeObjects, {
            Object = object,
            Property = property
        })
        pcall(function()
            object[property] = PresetColor
        end)
    end
end

local function RegisterBorder(object, thickness)
    if not object then
        return
    end

    local stroke = object:FindFirstChild("__UIBorder")

    if not stroke then
        stroke = Instance.new("UIStroke")
        stroke.Name = "__UIBorder"
        stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
        stroke.LineJoinMode = Enum.LineJoinMode.Round
        stroke.Thickness = thickness or 1.5
        stroke.Transparency = 1
        stroke.Parent = object
    end

    stroke.Color = PresetColor

    table.insert(BorderObjects, {
        Object = object,
        Stroke = stroke
    })

    stroke.Transparency = BordersEnabled and 0 or 1
end

local function ApplyThemeColor(color, save)
    if typeof(color) ~= "Color3" then
        return
    end

    if save then
        ThemeColor = color
    end

    PresetColor = color

    for i = #ThemeObjects, 1, -1 do
        local data = ThemeObjects[i]
        local object = data.Object
        local property = data.Property

        if object and object.Parent then
            pcall(function()
                object[property] = color
            end)
        else
            table.remove(ThemeObjects, i)
        end
    end

    for i = #BorderObjects, 1, -1 do
        local data = BorderObjects[i]
        local object = data.Object
        local stroke = data.Stroke

        if object and object.Parent and stroke and stroke.Parent then
            stroke.Color = color
        else
            table.remove(BorderObjects, i)
        end
    end

    if ActiveTabTitle and ActiveTabTitle.Parent then
        ActiveTabTitle.TextColor3 = color
    end
end

local function SetBorders(state)
    BordersEnabled = state

    for i = #BorderObjects, 1, -1 do
        local data = BorderObjects[i]
        local object = data.Object
        local stroke = data.Stroke

        if object and object.Parent and stroke and stroke.Parent then
            stroke.Transparency = state and 0 or 1
            stroke.Color = PresetColor
        else
            table.remove(BorderObjects, i)
        end
    end
end

local function MakeDraggable(topbarobject, object)
    local dragging = false
    local dragInput
    local dragStart
    local startPosition

    local function update(input)
        if not dragging or not dragStart or not startPosition then
            return
        end

        local delta = input.Position - dragStart

        object.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end

    topbarobject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then

            dragging = true
            dragStart = input.Position
            startPosition = object.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    dragInput = nil
                end
            end)
        end
    end)

    topbarobject.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then

            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            update(input)
        end
    end)
end

                value = math.clamp(value, min, max)

                value = math.floor(value + 0.5)
                Value.Text = tostring(value)

                CurrentValueFrame.Size = UDim2.new(percent, 0, 0, 3)
                Circle.Position = UDim2.new(percent, 0, 0.5, 0)

                    task.spawn(callback, value)

                local width = SlideFrame.AbsoluteSize.X

                if width <= 0 then
                    return
                end

                local percent = math.clamp(
                    (input.Position.X - SlideFrame.AbsolutePosition.X) / width,
                    0,
                    1
                )

                setValue(
                    min + ((max - min) * percent),
                    true
                )
            end

                if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then

                    dragging = true
                    update(input)
                end
            end)

            UserInputService.InputChanged:Connect(function(input)
                if dragging and (
                    input.UserInputType == Enum.UserInputType.MouseMovement
                    or input.UserInputType == Enum.UserInputType.Touch

                end
            end)

                if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                    dragging = false
                end
            end)

            setValue(start, false)
        end

                    (input.Position.Y - Color.AbsolutePosition.Y) /
                    Color.AbsoluteSize.Y,
                    0,
                    1
                )

                ColorS = x
                ColorV = 1 - y

                ColorSelection.Position = UDim2.new(x, 0, y, 0)

                updateColor()
            end

            local function updateHueFromInput(input)
                local y = math.clamp(
                    (input.Position.Y - Hue.AbsolutePosition.Y) /
                    Hue.AbsoluteSize.Y,
                    0,
                    1
                )

                ColorH = 1 - y

                HueSelection.Position = UDim2.new(
                    0.5,
                    0,
                    y,
                    0
                )

                updateColor()
            end

            Color.InputBegan:Connect(function(input)
                if rainbow then
                    return
                end

                if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then

                    colorDragging = true
                    updateColorFromInput(input)
                end
            end)

            Hue.InputBegan:Connect(function(input)
                if rainbow then
                    return
                end

                if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then

                    hueDragging = true
                    updateHueFromInput(input)
                end
            end)

            UserInputService.InputChanged:Connect(function(input)
                if rainbow then
                    return
                end

                if colorDragging and (
                    input.UserInputType == Enum.UserInputType.MouseMovement
                    or input.UserInputType == Enum.UserInputType.Touch
                ) then
                    updateColorFromInput(input)
                end

                if hueDragging and (
                    input.UserInputType == Enum.UserInputType.MouseMovement
                    or input.UserInputType == Enum.UserInputType.Touch
                ) then
                    updateHueFromInput(input)
                end
            end)

            UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then

                    colorDragging = false
                    hueDragging = false
                end
            end)

            Rainbow.Activated:Connect(function()
                if destroyed then
                    return
                end

                rainbow = not rainbow
                Rainbow.Text = rainbow and "Rainbow: ON" or "Rainbow: OFF"

                if rainbow then
                    if rainbowConnection then
                        rainbowConnection:Disconnect()
                    end

                    rainbowConnection = RunService.RenderStepped:Connect(function()
                        if destroyed or not Colorpicker.Parent then
                            if rainbowConnection then
                                rainbowConnection:Disconnect()
                                rainbowConnection = nil
                            end
                            return
                        end

                        local color = Color3.fromHSV(
                            lib.RainbowColorValue,
                            1,
                            1
                        )

                        BoxColor.BackgroundColor3 = color
                        Color.BackgroundColor3 = color

                        if typeof(callback) == "function" then
                            task.spawn(callback, color)
                        end
                    end)
                else
                    if rainbowConnection then
                        rainbowConnection:Disconnect()
                        rainbowConnection = nil
                    end

                    updateColor()
                end
            end)

            Confirm.Activated:Connect(function()
                colorPickerToggled = false
                Color.Visible = false
                Hue.Visible = false
                Confirm.Visible = false
                Rainbow.Visible = false

                Colorpicker:TweenSize(
                    UDim2.fromOffset(363, 42),
                    Enum.EasingDirection.Out,
                    Enum.EasingStyle.Quart,
                    0.2,
                    true
                )
            end)

            updateColor()
        end

        function tabcontent:Bind(bindText, keypreset, callback)
            local binding = false
            local key = keypreset or Enum.KeyCode.Unknown

            local Bind = Instance.new("TextButton")
            Bind.Name = "Bind"
            Bind.Parent = Tab
            Bind.BackgroundColor3 = Color3.fromRGB(34, 34, 34)
            Bind.Size = UDim2.fromOffset(363, 42)
            Bind.AutoButtonColor = false
            Bind.Text = ""
            Bind.ZIndex = 3

            local Corner = Instance.new("UICorner")
            Corner.CornerRadius = UDim.new(0, 8)
            Corner.Parent = Bind

            RegisterBorder(Bind, 1.4)

            local BindTitle = Instance.new("TextLabel")
            BindTitle.Parent = Bind
            BindTitle.BackgroundTransparency = 1
            BindTitle.Position = UDim2.new(0.0358, 0, 0, 0)
            BindTitle.Size = UDim2.new(0.55, 0, 1, 0)
            BindTitle.Font = Enum.Font.Gotham
            BindTitle.Text = bindText or "Bind"
            BindTitle.TextColor3 = Color3.new(1, 1, 1)
            BindTitle.TextSize = 14
            BindTitle.TextXAlignment = Enum.TextXAlignment.Left

            local KeyText = Instance.new("TextLabel")
            KeyText.Parent = Bind
            KeyText.BackgroundTransparency = 1
            KeyText.Position = UDim2.new(0.5, 0, 0, 0)
            KeyText.Size = UDim2.new(0.46, 0, 1, 0)
            KeyText.Font = Enum.Font.Gotham
            KeyText.Text = key.Name
            KeyText.TextColor3 = Color3.new(1, 1, 1)
            KeyText.TextSize = 14
            KeyText.TextXAlignment = Enum.TextXAlignment.Right

            Bind.Activated:Connect(function()
                if binding or destroyed then
                    return
                end

                binding = true
                KeyText.Text = "..."

                local connection

                connection = UserInputService.InputBegan:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.Keyboard then
                        key = input.KeyCode
                        KeyText.Text = key.Name
                        binding = false

                        if connection then
                            connection:Disconnect()
                            connection = nil
                        end
                    end
                end)
            end)

            UserInputService.InputBegan:Connect(function(input, processed)
                if processed or binding or destroyed then
                    return
                end

                if input.KeyCode == key
                    and key ~= Enum.KeyCode.Unknown then

                    if typeof(callback) == "function" then
                        task.spawn(callback)
                    end
                end
            end)
        end

        return tabcontent
    end
